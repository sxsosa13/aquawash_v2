marco3.pack(fill="both", expand=True) bal = ttk.LabelFrame(fin, text="Balance económico", padding=6) bal.pack(side="left", fill="both", expand=True, padx=(10, 0)) self.balance_var = tk.StringVar(value="") ttk.Label(bal, textvariable=self.balance_var, justify="left", font=("Segoe UI", 11)).pack(anchor="w", padx=8, pady=12) self.generar() def generar(self): try: d, h = db.ingresos_por_dia(d, h) self.egresos = db.egresos_por_dia(d, h) self.stock = db.insumos_bajo_stock() self.balance = db.balance_economico(d, h) llenar(self.tv_ing, [(i, (r["dia"].isoformat(), graficar(self, datos): c = self.canvas c.delete("all") if not datos: c.create_text(190, 120, text="Sin datos en el período", fill="#777") return maximo = max(r["cantidad"] for r in datos) for i, r in enumerate(datos[:8]): y = 14 + i * 30 c.create_text(4, y + 11, anchor="w", text=r["nombre"][:24], font=("Segoe UI", 9)) ancho = int(180 * r["cantidad"] / maximo) c.create_rectangle(190, y, 190 + ancho, y + 20, fill="#0a66c2", outline="") c.create_text(196 + ancho, y + 10, anchor="w", text=str(r["cantidad"]), font=("Segoe UI", 9, "bold")) def exportar(self): if not self.ingresos and not self.egresos: return messagebox.showinfo("Sin datos", "Genere primero un reporte con datos.") ruta = filedialog.asksaveasfilename(defaultextension=".csv", filetypes=[("CSV", "*.csv")], initialfile="sosa_car_wash_reporte.csv") if ruta: with open(ruta, "w", newline="", encoding="utf-8-sig") as f: w = csv.writer(f) w.writerow(["tipo", "fecha", "total"]) for r in self.ingresos: w.writerow(["ingreso", r["dia"].isoformat(), r["total"]]) for r in self.egresos: w.writerow(["egreso", r["dia"].isoformat(), r["total"]]) messagebox.showinfo("Exportado", f"Archivo guardado en:\n{ruta}")
